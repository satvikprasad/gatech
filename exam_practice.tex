\begin{enumerate}
  \item
    \begin{enumerate}
      \item We use the principle of inclusion-exclusion, defining
        $n-1$ conditions. Then,
        \begin{align*}
          N(\bar c_1 \bar c_2 \ldots \bar c_n) &= \sum_{k =
          0}^{n-1}(-1)^k {n-1 \choose k} (n - k)! \\
          &= \sum_{k=0}^{n-1}(-1)^k \dfrac{(n-1)!}{k! }(n-k).
        \end{align*}

      \item We define $n$ conditions, counting linear permutations
        and then dividing by $n.$
    \end{enumerate}
\end{enumerate}
