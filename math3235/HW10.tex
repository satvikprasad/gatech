\begin{enumerate}
  \item A fair dice is thrown 12000 times, and let $S$ denote the number of sixes thrown. Define sample space $\Omega = \{1, 2, \ldots, 6\}^{12000}.$ For $1 \leq i \leq 12000,$ let $X_i : \Omega \to \mathbb R$ be defined such that
    \begin{align*}
      X_i(\omega)  =
      \begin{cases}
        1 & \omega_i = 6 \\
        0 & \text{otherwise}
      \end{cases}
    \end{align*}

    Then, $$S = \sum_{i = 1}^{12000} X_i,$$ and applying the CLT yields $S \approx 12000\mu + \sigma \sqrt{12000}Z = 2000 + 20\sigma \sqrt{30} Z.$

    Finding the variance of a single $X_i,$ we get
    $$
    \mathbb E[X_i^2] = 1/6 \Rightarrow \sigma^2 = 1/6 - (1/6)^2 = 5/36.
    $$

    Thus, we have $$
    f_S(x) \approx f_{2000 + \frac{50}{3} \sqrt{6} \cdot Z}(x),
    $$
    where $Z \sim \mathcal N(0, 1).$

    Hence, the probability satisfies
    \begin{align*}
      \mathbb P(1900 < S < 2200) \approx \mathbb P\left(-\sqrt{6} < Z < 2 \sqrt{6}\right) = \int_{-\sqrt{6}}^{2 \sqrt{6}} \dfrac{1}{\sqrt{2\pi}} e^{- \frac{1}{2}x^2} \, dx
    \end{align*}
    as required.

  \item
    \begin{enumerate}
      \item Suppose $X_N \to X$ in $L^2$, meaning we have some RV $X$ such that
        \begin{align*}
          \lim_{n \to \infty} \mathbb E[(X_n - X)^2] = 0.
        \end{align*}

        By Jensen's inequality, since $f: \mathbb R \to \mathbb R, \ f(x) = x^2$ is convex, we know
        $$
        \mathbb E[|X_n - X|^2] \geq \mathbb E[|X_n - X|]^2 \geq 0,
        $$
        since $|X_n - X|$ is always positive.

        Applying limits to both sides, by the squeeze theorem we must have $$
        \lim_{n \to \infty} \mathbb E[|X_n - X|]^2 = 0 \Rightarrow \lim_{n \to \infty} \mathbb E[|X_n - X|] = 0.
        $$

        Once again, by convexity of $f: \mathbb R \to \mathbb R^{+}, \ f(x) = |x|,$ we know
        $$
        \mathbb E[|X_n - X|] \geq |\mathbb E[X_n - X]| = |\mathbb E[X_n] - \mathbb E[X]| \geq 0,
        $$
        and by the squeeze theorem again,
        $$
        \lim_{n \to \infty} |\mathbb E[X_n] - \mathbb E[X]| = 0 \Rightarrow \mathbb \lim_{n \to \infty} E[X_n] = \mathbb E[X]
        $$
        as required.

      \item If $X_n \to X$ in probability, we know for every $\epsilon > 0,$ $$
        \lim_{n \to \infty} \mathbb P(|X_n - X| \geq \epsilon) = 0.
        $$

        Take counterexample $X_n = n$ with probability $1/n,$ and $0$ otherwise. Then, $\mathbb E[X_n] = 1,$ but $X_n$ converges to $0$ in probability.
    \end{enumerate}

  \item Let $X_1, X_2, \ldots$ be independent random variables, each having the uniform distribution on the interval $(0, a),$ and let $Z_n = \max\{X_1, X_2, \ldots, X_n\}.$

    \begin{enumerate}
      \item Finding the CDF for $Z_n,$ we know $$
        \mathbb P(Z_n \leq z) = \mathbb P(\max\{X_1, X_2, \ldots, X_{n}\} \leq z).
        $$
        Conditioning on $X_n,$ we have
        $$
        \mathbb P(Z_n \leq z) =  \dfrac{1}{a}\int_{0}^{a}\mathbb P(\max\{X_1, X_2, \ldots, X_{n-1}, t \}\leq z)\, dt.
        $$

        Then, partioning on $Z_{n -1} \leq t,$ we have
        $$
        \mathbb P(Z_n \leq z) = \dfrac{1}{a} \int_{0}^{a} \ind{t \leq z} \mathbb P(Z_{n-1} \leq t) + \mathbb P(t \leq Z_{n - 1} \leq z) \, dt.
        $$

        Letting $F_n(z) = \mathbb P(Z_n \leq z),$ we get
        \begin{align*}
          F_n(z) &= \dfrac{1}{a} \int_{0}^{z} F_{n - 1}(t) + F_{n - 1}(z) - F_{n-1}(t) \, dt \\
          &= \dfrac{z}{a} F_{n-1}(z)
        \end{align*}
        for $n \geq 2.$ When $n = 1$, $F_{1}(z) = \dfrac{z}{a}$ for $0 \leq z \leq a,$ $1$ above, and $0$ below. Therefore, we have $$
        F_n(z) =
        \begin{cases}
          \dfrac{z^n}{a^n} & 0 \leq z \leq a \\
          0 & z < 0 \\
          1 & z \geq a
        \end{cases} \Rightarrow f_{Z_n}(z) =
        \begin{cases}
          \dfrac{n}{a^n}z^{n-1} & 0 \leq z \leq a \\
          0 & \text{otherwise}
        \end{cases}.
        $$

        Hence, for $a \geq \epsilon > 0,$ we know
        \begin{align*}
          \mathbb P(|Z_n - a| \geq \epsilon) &= \mathbb P(\{Z_n \geq a + \epsilon\} \cup \{Z_n \leq a - \epsilon \})  \\
          &= 0 + \mathbb P(Z_n \leq a - \epsilon) \\
          &= \int_{0}^{a - \epsilon} \dfrac{n}{a^{n}} z^{n-1} \, dz \\
          &= \dfrac{(a - \epsilon)^n}{a^n} \\
          &= \left(1 - \dfrac{\epsilon}{a}\right)^n.
        \end{align*}

        Taking the limit as $n \to \infty$ yields
        \begin{align*}
          \lim_{n \to \infty} \mathbb P(|Z_n - a| \geq \epsilon) = \lim_{n \to \infty} \left(1 - \dfrac{\epsilon}{a}\right)^n  = 0
        \end{align*}
        whenever $\epsilon > 0,$ since $1 - \epsilon / a < 1.$ When $\epsilon > a,$ probability is always $0$ (trivial).

        Therefore, $Z_n \to a$ in probability as $n \to \infty.$

      \item Similarly for positive $0 < \epsilon \leq \sqrt{a}$ we know
        \begin{align*}
          \mathbb P(|\sqrt{Z_n} - \sqrt{a}| \geq \epsilon) &= \mathbb P(\sqrt{Z_n} \geq \sqrt{a} + \epsilon) + \mathbb P(\sqrt{Z_n} \leq \sqrt{a} - \epsilon) \\
          &= F_n((\sqrt{a} - \epsilon)^2) \\
          &= \dfrac{(\sqrt{a} - \epsilon)^{2n}}{a^n} \\
          &= \left (1 - \dfrac{\epsilon}{\sqrt{a}}\right)^{2n}.
        \end{align*}

        For $0 < \epsilon \leq \sqrt{a},$ the inside term is bounded between 0 (inclusive) and 1 (exclusive), meaning in the limit we have
        \begin{align*}
          \lim_{n \to \infty} \mathbb P(|\sqrt{Z_n} - \sqrt{a}| \geq \epsilon) = 0
        \end{align*}
        for all $\epsilon > 0.$ Hence, $\sqrt{Z_n} \to \sqrt{a}$ in probability as $n \to \infty.$
    \end{enumerate}

\end{enumerate}
